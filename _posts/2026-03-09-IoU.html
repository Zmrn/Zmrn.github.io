C#实现的Rotated IoU算法

分类: algorithm

实际使用中遇到了需要计算两个OBB（Oriented Bounding Box）交叠面积的需求，查找资料发现了相关算法：

Rotated IoU 计算

将其用C#实现并学习了实现方法。代码放在文章最后。

基本原理

该算法通过构造两个OBB交叠区域的多边形，来计算该多边形的面积。 示例图中用填充白色表示两个矩形OBB；用红色Gizmos线围成交叠多边形；用红色Gizmos小球表示交叠多边形的顶点。小球的大小代表顶点的顺序，最小的是第一个顶点，最大的是最后一个顶点。

  1. 计算交叠多边形的顶点
    1. 任一OBB的一个顶点如果进入了另一个OBB的内部，那么这个顶点是交叠多边形的一个顶点。
    2. 任一OBB的一条边如果和另一个OBB的任意条边有交点，那么这些交点是交叠多边形的一个顶点。
  2. 对多边形顶点进行排序，以构成多边形
    1. 计算该多边形的几何中心。
    2. 围绕几何中心，对多边形的顶点进行归一化。
    3. 对归一化的向量按照顺时针排序（逆时针也行），归一化后的向量坐落在球面上，根据y轴正负，按照x轴的向量值进行排序即可。
  3. 计算该多边形的面积
    1. 计算构成该多边形的三角形面积的和即可。
    2. 由于该图形必为凸多边形，选择第一个节点，并顺序两两遍历其他节点，即可得到构成该多边形的三角形。如下图的多边形由三个三角形构成。

using System.Collections.Generic;
using UnityEngine;
using System;

public static class CalculateRotateIoU 
{
    /// <summary>
    /// 一个OBB的基本数据
    /// </summary>
    public struct OBBData
    {
        public Vector2 Position;
        
        public Vector2 Size;

        public float Angle;

        public float Area
        {
            get
            {
                return Size.x * Size.y;
            }
        }

        public void DebugPrint()
        {
            Debug.Log($"Position: {Position}, Size: {Size}, Angle: {Angle}");
        }
    }
    
    /// <summary>
    /// 记录一个OBB顺时针方向的角点
    /// </summary>
    struct BoxCornerPoints
    {
        public float X0, Y0, X1, Y1, X2, Y2, X3, Y3;
        
        public float this[int index]
        {
            get
            {
                switch (index)
                {
                    case 0:
                        return this.X0;
                    case 1:
                        return this.Y0;
                    case 2:
                        return this.X1;
                    case 3:
                        return this.Y1;
                    case 4:
                        return this.X2;
                    case 5:
                        return this.Y2;
                    case 6:
                        return this.X3;
                    case 7:
                        return this.Y3;
                    default:
                        throw new IndexOutOfRangeException("Invalid BoxCornerPoints index!");
                }
            }
            set
            {
                switch (index)
                {
                    case 0:
                        this.X0 = value;
                        break;
                    case 1:
                        this.Y0 = value;
                        break;
                    case 2:
                        this.X1 = value;
                        break;
                    case 3:
                        this.Y1 = value;
                        break;
                    case 4:
                        this.X2 = value;
                        break;
                    case 5:
                        this.Y2 = value;
                        break;
                    case 6:
                        this.X3 = value;
                        break;
                    case 7:
                        this.Y3 = value;
                        break;
                    default:
                        throw new IndexOutOfRangeException("Invalid BoxCornerPoints index!");
                }
            }
        }
    }
    
    // 保存交叠区域的角点
    public static List<Vector2> Points = new List<Vector2>(8);

    // 保存交叠区域的中点
    static Vector2 UnionCenter = Vector2.zero;
    
    /// <summary>
    /// 将obb数据转化为角点
    /// </summary>
    /// <returns>角点数据</returns>
    static BoxCornerPoints Box2Corners(OBBData box1)
    {
        var x = box1.Position.x;
        var y = box1.Position.y;
        var width = box1.Size.x;
        var height = box1.Size.y;
        
        var cos = Mathf.Cos(box1.Angle);
        var sin = Mathf.Sin(box1.Angle);
        
        var halfWidth = width * 0.5f;
        var halfHeight = height * 0.5f;
        var cornersX = new Vector4(-halfWidth, -halfWidth, halfWidth, halfWidth);
        var cornersY = new Vector4(-halfHeight, halfHeight, halfHeight, -halfHeight);
            
        var corners = new BoxCornerPoints();
        for (var i = 0; i < 4; i++)
        {
            corners[2 * i] = cos * cornersX[i] + sin * cornersY[i] + x;
            corners[2 * i + 1] = -sin * cornersX[i] + cos * cornersY[i] + y;
        }
        return corners;
    }
    
    /// <summary>
    /// 判断一个点是否在OBB的内部
    /// </summary>
    static bool IsPointInQuad(Vector2 point, BoxCornerPoints corners)
    {
        var ab0 = corners[2] - corners[0];
        var ab1 = corners[3] - corners[1];
        
        var ad0 = corners[6] - corners[0];
        var ad1 = corners[7] - corners[1];
        
        var ap0 = point.x - corners[0];
        var ap1 = point.y - corners[1];
        
        var abab = ab0 * ab0 + ab1 * ab1;
        var abap = ab0 * ap0 + ab1 * ap1;
        var adad = ad0 * ad0 + ad1 * ad1;
        var adap = ad0 * ap0 + ad1 * ap1;
        
        return abab >= abap && abap >= 0 && adad >= adap && adap >= 0;
    }
    
    /// <summary>
    /// 计算线段的交点
    /// </summary>
    /// <param name="pts1">OBB1角点</param>
    /// <param name="pts2">OBB2角点</param>
    /// <param name="i">OBB1的第几条边</param>
    /// <param name="j">OBB2的第几条边</param>
    /// <returns>(是否有交点, 交点)</returns>
    static (bool, Vector2) LineSegmentIntersection(BoxCornerPoints pts1, BoxCornerPoints pts2, int i, int j)
    {
        var A = Vector2.zero;
        var B = Vector2.zero;
        var C = Vector2.zero;
        var D = Vector2.zero;
        var ret = Vector2.zero;
        
        A[0] = pts1[2 * i];
        A[1] = pts1[2 * i + 1];

        B[0] = pts1[2 * ((i + 1) % 4)];
        B[1] = pts1[2 * ((i + 1) % 4) + 1];

        C[0] = pts2[2 * j];
        C[1] = pts2[2 * j + 1];

        D[0] = pts2[2 * ((j + 1) % 4)];
        D[1] = pts2[2 * ((j + 1) % 4) + 1];
        
        var BA0 = B[0] - A[0];
        var BA1 = B[1] - A[1];
        var DA0 = D[0] - A[0];
        var CA0 = C[0] - A[0];
        var DA1 = D[1] - A[1];
        var CA1 = C[1] - A[1];

        var acd = DA1 * CA0 > CA1 * DA0;
        var bcd = (D[1] - B[1]) * (C[0] - B[0]) > (C[1] - B[1]) * (D[0] - B[0]);
        if (acd != bcd)
        {
            var abc = CA1 * BA0 > BA1 * CA0;
            var abd = DA1 * BA0 > BA1 * DA0;
            if (abc != abd)
            {
                var DC0 = D[0] - C[0];
                var DC1 = D[1] - C[1];
                var ABBA = A[0] * B[1] - B[0] * A[1];
                var CDDC = C[0] * D[1] - D[0] * C[1];
                var DH = BA1 * DC0 - BA0 * DC1;
                var Dx = ABBA * DC0 - BA0 * CDDC;
                var Dy = ABBA * DC1 - BA1 * CDDC;
                ret[0] = Dx / DH;
                ret[1] = Dy / DH;
                return (true, ret);
            }
        }
        return (false, ret);
    }
    
    /// <summary>
    /// 将向量列表根据顺时针排序
    /// </summary>
    static void SortVertexInConvexPolygon(List<Vector2> points)
    {
        // 用来给点排序的算法，依据点到交叠区域中心的距离排序
        void QuickSortForPoints(List<Vector2> pointList)
        {
            // 将向量列表归一化后，按照x轴排序
            float GetNormalizedPointDistanceToCenter(Vector2 point)
            {
                var vx = point[0] - UnionCenter[0];
                var vy = point[1] - UnionCenter[1];
                var d = Mathf.Sqrt(vx * vx + vy * vy); // 需要归一化，似乎无法避免这个开根号
                vx /= d;
                vy /= d;
                if (vy < 0)
                {
                    vx = -2 - vx;
                }
                return vx;
            }
            
            // 一个快速排序算法
            void QuickSortForPointsInternal(List<Vector2> pointList, int left, int right)
            {
                if (left < right)
                {
                    int i = left;
                    int j = right;
                    var middle = pointList[(left + right) / 2];
                    while (true)
                    {
                        while (i < right && GetNormalizedPointDistanceToCenter(pointList[i]) < GetNormalizedPointDistanceToCenter(middle)) { i++; };
                        while (j > 0 && GetNormalizedPointDistanceToCenter(pointList[j]) > GetNormalizedPointDistanceToCenter(middle)) { j--; };
                        if (i == j) break;
                        (pointList[i], pointList[j]) = (pointList[j], pointList[i]);
                        if (pointList[i] == pointList[j]) j--;
                    }
                    QuickSortForPointsInternal(pointList, left, i);
                    QuickSortForPointsInternal(pointList, i + 1, right);
                }
            }
        
            QuickSortForPointsInternal(pointList, 0, pointList.Count - 1);
        }

        if (points.Count <= 0)
        {
            return;
        }
        
        UnionCenter = Vector2.zero;

        for (int i = 0; i < points.Count; i++)
        {
            UnionCenter[0] += points[i][0];
            UnionCenter[1] += points[i][1];
        }
        UnionCenter[0] /= points.Count;
        UnionCenter[1] /= points.Count;

        QuickSortForPoints(points);
    }
    
    /// <summary>
    /// 将多边形转化为多个三角形面积之和
    /// </summary>
    /// <param name="points">输入多边形的顶点</param>
    /// <returns>多边形的面积</returns>
    static float Area(List<Vector2> points)
    {
        float TrangleArea(Vector2 a, Vector2 b, Vector2 c)
        {
            return ((a[0] - c[0]) * (b[1] - c[1]) - (a[1] - c[1]) * (b[0] - c[0])) * 0.5f;
        }

        var areaValue = 0f;
        for (int i = 0; i < points.Count - 2; i++)
        {
            areaValue += Mathf.Abs(TrangleArea(points[0], points[i + 1], points[i + 2]));
        }
        return areaValue;
    }
    
    /// <summary>
    /// 计算两个OBB的交叠面积
    /// </summary>
    public static float IoU(OBBData box1, OBBData box2)
    {
        // 初始化缓存，这个列表长度最多为8，已初始化为长度8
        Points.Clear();
        
        // 转化成角点
        var corners1 = Box2Corners(box1);
        var corners2 = Box2Corners(box2);
        
        // 如果OBB1的一个点在OBB2中，那么它肯定是一个构成交叠区域的角点
        for (var i = 0; i < 4; i++)
        {
            var point = new Vector2(corners1[2 * i], corners1[2 * i + 1]);
            if (IsPointInQuad(point, corners2))
            {
                Points.Add(point);
            }
        }

        // 如果OBB2的一个点在OBB1中，那么它肯定是一个构成交叠区域的角点
        for (var i = 0; i < 4; i++)
        {
            var point = new Vector2(corners2[2 * i], corners2[2 * i + 1]);
            if (IsPointInQuad(point, corners1))
            {
                Points.Add(point);
            }
        }

        // 计算OBB1和OBB2的线段交点
        for (var i = 0; i < 4; i++)
        {
            for (int j = 0; j < 4; j++)
            {
                var (ret, point) = LineSegmentIntersection(corners1, corners2, i, j);
                if (ret)
                {
                    Points.Add(point);
                }
            }
        }
        
        // 排序交叠区域的顶点后计算面积
        SortVertexInConvexPolygon(Points);
        var result = Area(Points);
        return result;
    }
}
